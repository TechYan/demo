// Recipe: write AA/55 data with alternating byte enables; untouched bytes stay initialized.
class svip_axi_patterns_recipe_seq extends svip_axi_generation_seq;
  `uvm_object_utils(svip_axi_patterns_recipe_seq)
  function new(string name="svip_axi_patterns_recipe_seq"); super.new(name); endfunction
  virtual task body();
    svip_axi_addr_t address=64'h1234567800001000;
    byte unsigned old_bytes[svip_axi_addr_t];
    svip_axi_txn t,r;
    svip_axi_generation_goal_t goal;
    svip_axi_strb_e modes[3]='{SVIP_AXI_STRB_FULL,SVIP_AXI_STRB_RANDOM,SVIP_AXI_STRB_MIXED};
    int bytes=p_sequencer.cfg.data_bytes;
    wait(ctrl.resetn===1); repeat(5) @(negedge ctrl.clk);
    write_check(address,1,$clog2(bytes));
    old_bytes=oracle; t=request(0,address,1,$clog2(bytes)); oracle=old_bytes;
    for(int lane=0;lane<bytes;lane++) begin
      t.data[0][lane*8+:8]=lane%2 ? 8'h55 : 8'haa;
      t.strb[0][lane]=(lane%2)==0;
      if((lane%2)==0) oracle[address+lane]=8'haa;
    end
    completed(t,r); read_check(address,1,$clog2(bytes));
    // Public helper modes: initialize untouched bytes, send the filled object, check readback.
    foreach(modes[m]) begin
      goal=make_goal(SVIP_AXI_WR,address+'h1000,2,$clog2(bytes),SVIP_AXI_INCR,'h710+m);
      write_check(goal.addr,goal.beats,goal.size,goal.burst,91);
      t=randomized_request(goal);
      t.fill_wdata(modes[m]);
      send_generated(t,goal,modes[m]);
    end
  endtask

// Recipe: a sequence chooses when the subordinate processes and replies to requests.
class svip_axi_slave_model_recipe_seq extends svip_axi_contract_seq;
  `uvm_object_utils(svip_axi_slave_model_recipe_seq)
  function new(string name="svip_axi_slave_model_recipe_seq"); super.new(name); endfunction
  virtual task body();
    int size,reply_count;
    wait(ctrl.resetn===1); repeat(5) @(negedge ctrl.clk);
    size=p_sequencer.cfg.max_size()<3 ? p_sequencer.cfg.max_size() : 3;
    if(sub.cfg.memory_mode!=SVIP_AXI_MEMORY || sub.cfg.mem==null)
      `uvm_fatal("SLAVE_RECIPE_MEMORY","default internal memory was not installed")
    fork
      begin
        svip_axi_txn t,r;
        t=request(0,'h1000,2,size,SVIP_AXI_INCR,1,37); completed(t,r);
        if(r.buser!=svip_axi_user_t'('h31)) `uvm_error("SLAVE_RECIPE_BUSER","sequence-selected BUSER was lost")
        t=request(1,'h1000,2,size,SVIP_AXI_INCR,1); completed(t,r);
        if(r.rresp.size()!=2 || r.ruser.size()!=2) `uvm_fatal("SLAVE_RECIPE_READ","read response omitted beats")
        foreach(r.rresp[i])
          if(r.rresp[i]!=SVIP_AXI_OKAY || r.ruser[i]!=svip_axi_user_t'('h51+i))
            `uvm_error("SLAVE_RECIPE_RUSER","sequence-selected read status/USER was lost")
      end
      begin
        svip_axi_txn req,rsp;
        repeat(2) begin
          sub.get_request(req);
          repeat(3) @(negedge ctrl.clk);
          // This explicit call commits a write or snapshots a read exactly once.
          rsp=sub.default_response(req);
          if(rsp==null) `uvm_fatal("SLAVE_RECIPE_DEFAULT","default processing rejected a live request")
          if(req.dir==SVIP_AXI_WR) rsp.buser=svip_axi_user_t'('h31);
          else foreach(rsp.ruser[i]) rsp.ruser[i]=svip_axi_user_t'('h51+i);
          if(!sub.send_response(req,rsp)) `uvm_fatal("SLAVE_RECIPE_SEND","live response was rejected")
          reply_count++;
        end
      end
    join
    wait_all(); repeat(8) @(negedge ctrl.clk);
    if(n_sent!=2 || n_done!=2 || reply_count!=2)
      `uvm_error("SLAVE_RECIPE_COMPLETIONS","manager and responder did not complete exactly two requests")
  endtask
endclass

class svip_apb_backpressure_test extends svip_apb_base_test;
  `uvm_component_utils(svip_apb_backpressure_test)
  function new(string name,uvm_component parent);super.new(name,parent);endfunction
  task stalled(svip_apb_data_t expected,int cycles);
    svip_apb_strb_t expected_strb=0;
    if(cfg.apb4)for(int lane=0;lane<cfg.data_width/8;lane++)expected_strb[lane]=1;
    do @(vif.mon_cb);while(vif.mon_cb.psel!==1 || vif.mon_cb.penable!==1);
    for(int i=0;i<cycles;i++)begin
      if(vif.mon_cb.pready!==0 || vif.mon_cb.psel!==1 || vif.mon_cb.penable!==1 ||
         vif.mon_cb.paddr!=='h80 || vif.mon_cb.pwrite!==1 || vif.mon_cb.pwdata!==expected ||
         vif.mon_cb.pprot!==0 || vif.mon_cb.pstrb!==expected_strb)
        `uvm_fatal("TEST_BACKPRESSURE","selected write changed while response was withheld")
      #1ps;
      if(ctrl.bus_completions!=1 || terminal.items.size()!=1)
        `uvm_fatal("TEST_BACKPRESSURE","request completed before PREADY was released")
      if(i+1<cycles)@(vif.mon_cb);
    end
  endtask
  task run_case();
    svip_apb_txn t=cfg.create_txn("blocked_write");
    svip_apb_data_t next_data=data_pattern('h2468ace0);
    bit abort_write=(case_name=="backpressure_reset_write");
    checked_write('h80,'h13579bdf);
    t.write=1;t.addr='h80;t.data=next_data;t.strb='1;
    ctrl.manual_response=1;ctrl.manual_ready=0;ctrl.manual_error=0;
    fork
      submit(t);
      begin
        stalled(next_data,16);
        if(abort_write)begin
          @(negedge ctrl.clk);ctrl.resetn=0;
          #1ps;
          if(vif.psel!==0 || vif.penable!==0) `uvm_fatal("TEST_BACKPRESSURE","reset did not idle pins")
          @(negedge ctrl.clk);ctrl.resetn=1;
        end else ctrl.manual_ready=1;
      end
    join
    if(t.status!==(abort_write?SVIP_APB_RESET_ABORTED:SVIP_APB_OK) || t.wait_cycles!=16)
      `uvm_fatal("TEST_BACKPRESSURE","held request lost its result or exact wait count")
    ctrl.manual_response=0;ctrl.manual_ready=1;
    if(!abort_write)for(int lane=0;lane<cfg.data_width/8;lane++)gold['h80+lane]=next_data[lane*8+:8];
    // This fixture commits memory on the actual bus handshake, never on a stall.
    checked_read('h80);
    if(ctrl.bus_completions!=(abort_write?2:3) || observed.items.size()!=(abort_write?2:3) || terminal.items.size()!=3)
      `uvm_fatal("TEST_BACKPRESSURE","false completion or lost recovery")
  endtask
